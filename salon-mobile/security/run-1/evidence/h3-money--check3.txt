import { describe, expect, it } from "vitest";
import * as E from "/home/user/Adds/salon-mobile/src/domain/engine";
import { emptyData, type Data } from "/home/user/Adds/salon-mobile/src/domain/types";

const SECRET = { salt: "x", hash: "y" };
function setup() {
  let data: Data = emptyData();
  const run = <T,>(op: { data: Data; result: T }) => ((data = op.data), op.result);
  const stylist = run(E.addStaff(data, { name: "Dummy", phone: "9811100001", chair: "Chair 1", secret: SECRET }));
  const c = run(E.createCustomer(data, { phone: "9876500000", name: "Dummy Cust", consent: true }));
  const pay = (price: number, wallet: number, codeOk?: boolean) => {
    const v = run(E.startVisit(data, { customerId: c.id, serviceIds: ["haircut"], addOnIds: [], staffId: stylist.id }));
    run(E.addVisitExtra(data, { visitId: v.id, name: "Dummy", price }));
    run(E.approveVisitExtras(data, v.id));
    run(E.updateOpenVisit(data, { visitId: v.id, serviceIds: [], addOnIds: [], staffId: stylist.id }));
    return run(E.checkoutVisit(data, { visitId: v.id, walletAmount: wallet, paymentMethod: "CASH", walletCodeOk: codeOk }));
  };
  return { get data() { return data; }, run, c, pay };
}

describe("h3-money", () => {
  it("CHECK-A closeWallet refunds bonus credit as cash after an earlier full spend", () => {
    const s = setup();
    s.run(E.topUp(s.data, { customerId: s.c.id, amount: 2000, method: "CASH" }));
    s.pay(2200, 2200, true);
    s.run(E.topUp(s.data, { customerId: s.c.id, amount: 2000, method: "CASH" }));

    console.log("CHECK-A closeWallet", JSON.stringify(r), "cash deposited in 2nd top-up = 2000");
    expect(r.refunded).toBe(2200);
  });
  it("CHECK-B wallet code threshold is per visit: split spends need no code", () => {
    const s = setup();
